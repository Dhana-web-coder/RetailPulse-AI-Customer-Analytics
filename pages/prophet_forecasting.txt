import streamlit as st
import pandas as pd
from prophet import Prophet
from sklearn.metrics import mean_absolute_error
import matplotlib.pyplot as plt

st.set_page_config(
    page_title="RetailPulse - Prophet Forecast",
    layout="wide"
)

st.title("RetailPulse - Sales Forecasting")
st.write("Daily sales forecasting using Facebook Prophet.")

# Load dataset
@st.cache_data
def load_data():
    df = pd.read_csv(
        "data/retail_2009_2010.csv",
        low_memory=False
    )

    df["InvoiceDate"] = pd.to_datetime(
        df["InvoiceDate"], errors="coerce"
    )

    df["Sales"] = df["Quantity"] * df["Price"]

    df = df.dropna(subset=["InvoiceDate", "Sales"])
    df = df[(df["Quantity"] > 0) & (df["Price"] > 0)]

    daily_sales = (
        df.set_index("InvoiceDate")["Sales"]
        .resample("D")
        .sum()
        .reset_index()
    )

    daily_sales.columns = ["ds", "y"]
    return daily_sales


try:
    daily_sales = load_data()

    st.subheader("Historical Daily Sales")
    st.line_chart(
        daily_sales.set_index("ds")["y"],
        x_label="Date",
        y_label="Sales"
    )

    forecast_days = st.slider(
        "Forecast days",
        min_value=7,
        max_value=90,
        value=30
    )

    if st.button("Generate Forecast"):
        with st.spinner("Training Prophet model..."):
            model = Prophet()
            model.fit(daily_sales)

            future = model.make_future_dataframe(
                periods=forecast_days,
                freq="D"
            )

            forecast = model.predict(future)

        st.success("Forecast generated successfully!")

        st.subheader("Sales Forecast Chart")

        fig = model.plot(forecast)
        st.pyplot(fig)

        st.subheader("Forecast Values")

        future_forecast = forecast[
            ["ds", "yhat", "yhat_lower", "yhat_upper"]
        ].tail(forecast_days)

        future_forecast.columns = [
            "Date",
            "Predicted Sales",
            "Lower Estimate",
            "Upper Estimate"
        ]

        st.dataframe(future_forecast, hide_index=True)

        st.metric(
            "Average Predicted Daily Sales",
            f"{future_forecast['Predicted Sales'].mean():,.2f}"
        )

except Exception as e:
    st.error(f"Forecasting error: {e}")
