import streamlit as st
import pandas as pd
import numpy as np
import plotly.express as px
from pathlib import Path
st.set_page_config(page_title="Forecast Evaluation", layout="wide")
st.title("RetailPulse - Forecast Model Evaluation")
BASE_DIR = Path(__file__).resolve().parent.parent
file_path = BASE_DIR / "data" / "forecast_evaluation_results.csv"
if not file_path.exists():
    st.error("Evaluation CSV file not found.")
    st.stop()
df = pd.read_csv(file_path)
required = ["Date", "ActualSales", "PredictedSales"]
if not all(c in df.columns for c in required):
    st.error(f"Missing columns. Found: {df.columns.tolist()}")
    st.stop()
df["Date"] = pd.to_datetime(df["Date"], errors="coerce")
for col in ["ActualSales", "PredictedSales"]:
    df[col] = pd.to_numeric(df[col], errors="coerce")
df = df.dropna(subset=required).sort_values("Date")
if df.empty:
    st.error("No valid evaluation data.")
    st.stop()
actual = df["ActualSales"].to_numpy()
predicted = df["PredictedSales"].to_numpy()
mae = np.mean(np.abs(actual - predicted))
rmse = np.sqrt(np.mean((actual - predicted) ** 2))
mask = actual != 0
mape = np.mean(np.abs((actual[mask] - predicted[mask]) / actual[mask])) * 100 if mask.any() else np.nan
c1, c2, c3, c4 = st.columns(4)
c1.metric("MAE", f"{mae:,.2f}")
c2.metric("RMSE", f"{rmse:,.2f}")
c3.metric("MAPE", f"{mape:.2f}%" if np.isfinite(mape) else "N/A")
c4.metric("Test Days", str(len(df)))
st.subheader("Actual vs Predicted Sales")
fig = px.line(df, x="Date", y=["ActualSales", "PredictedSales"], markers=True)
st.plotly_chart(fig, use_container_width=True)
df["AbsoluteError"] = (df["ActualSales"] - df["PredictedSales"]).abs()
st.subheader("Daily Prediction Error")
fig2 = px.bar(df, x="Date", y="AbsoluteError")
st.plotly_chart(fig2, use_container_width=True)
st.subheader("Evaluation Details")
st.dataframe(df, use_container_width=True)
st.download_button(
    "Download Evaluation Results",
    df.to_csv(index=False).encode("utf-8"),
    "forecast_evaluation_results.csv",
    "text/csv"
)
